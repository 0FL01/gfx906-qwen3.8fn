cmake_minimum_required(VERSION 3.25)
project(qwen_gfx906 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CORE_WITH_HIP "Build the gfx906 GPU executable" ON)
set(CORE_REVISION "unknown" CACHE STRING "Source Git revision")
option(CORE_DIRTY "Source has uncommitted changes" OFF)
include(CTest)
add_library(qwen-core STATIC src/model.cpp src/quant.cpp src/kv.cpp src/qsa.cpp)
target_include_directories(qwen-core PUBLIC src)
target_compile_options(qwen-core PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
add_executable(core-inspect src/inspect.cpp)
target_link_libraries(core-inspect PRIVATE qwen-core)
target_compile_options(core-inspect PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME baseline-client COMMAND ${Python3_EXECUTABLE} -B -m unittest discover
    -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p baseline_test.py)
  add_executable(model-test tests/model_test.cpp)
  target_link_libraries(model-test PRIVATE qwen-core)
  target_compile_options(model-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME model-loader COMMAND model-test)
  add_executable(quant-test tests/quant_test.cpp)
  target_link_libraries(quant-test PRIVATE qwen-core)
  target_compile_options(quant-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME quant-primitives COMMAND quant-test)
  add_executable(kv-test tests/kv_test.cpp)
  target_link_libraries(kv-test PRIVATE qwen-core)
  target_compile_options(kv-test PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
  add_test(NAME kv-primitives COMMAND kv-test)
  add_executable(qsa-test tests/qsa_test.cpp)
  target_link_libraries(qsa-test PRIVATE qwen-core)
  target_compile_options(qsa-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME qsa-selection COMMAND qsa-test)
  add_test(NAME kv-results COMMAND ${Python3_EXECUTABLE} -B -m unittest discover
    -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p kv_results_test.py)
  add_test(NAME expert-results COMMAND ${Python3_EXECUTABLE} -B -m unittest discover
    -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p expert_results_test.py)
endif()
if(CORE_WITH_HIP)
  set(CMAKE_HIP_ARCHITECTURES gfx906)
  set(GPU_TARGETS gfx906)
  enable_language(HIP)
  set(CMAKE_HIP_STANDARD 20)
  set(CMAKE_HIP_STANDARD_REQUIRED ON)
  find_package(rocblas REQUIRED CONFIG)
  add_library(qwen-kv-gpu STATIC src/hip/kv.hip)
  target_link_libraries(qwen-kv-gpu PUBLIC qwen-core)
  target_compile_options(qwen-kv-gpu PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  add_executable(core-kv src/kv_main.hip)
  target_link_libraries(core-kv PRIVATE qwen-kv-gpu)
  target_compile_definitions(core-kv PRIVATE CORE_REVISION="${CORE_REVISION}" CORE_DIRTY=$<BOOL:${CORE_DIRTY}>)
  target_compile_options(core-kv PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  add_executable(core-probe src/probe.cpp src/hip/probe.hip)
  target_include_directories(core-probe PRIVATE src)
  target_link_libraries(core-probe PRIVATE roc::rocblas)
  target_compile_definitions(core-probe PRIVATE CORE_REVISION="${CORE_REVISION}" CORE_DIRTY=$<BOOL:${CORE_DIRTY}>)
  target_compile_options(core-probe PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>)
  target_compile_options(core-probe PRIVATE $<$<COMPILE_LANGUAGE:HIP>:-Wall;-Wextra;-Werror>)
  add_executable(core-expert src/expert_main.cpp src/hip/expert.hip)
  target_link_libraries(core-expert PRIVATE qwen-core)
  target_compile_definitions(core-expert PRIVATE CORE_REVISION="${CORE_REVISION}" CORE_DIRTY=$<BOOL:${CORE_DIRTY}>)
  target_compile_options(core-expert PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>)
  target_compile_options(core-expert PRIVATE $<$<COMPILE_LANGUAGE:HIP>:-Wall;-Wextra;-Werror>)
endif()
